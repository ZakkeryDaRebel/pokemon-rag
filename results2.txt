WITH offensive AS (
    SELECT
        p.name AS offensive_pokemon,
        ROW_NUMBER() OVER (ORDER BY AVG(pokemon_type_effectiveness(m.type, mp.type1, mp.type2)) DESC) AS rn
    FROM trainer t
    JOIN pc ON t.trainer_id = pc.trainer_id
    JOIN pokedex p ON pc.pokemon_id = p.pokemon_id
    JOIN moves m ON m.move_id IN (pc.move1, pc.move2, pc.move3, pc.move4)
    JOIN pokedex mp ON mp.name IN ('Kingdra', 'Starmie')
    WHERE t.name = 'Red'
      AND m.power IS NOT NULL
    GROUP BY pc.catch_id, p.name
    LIMIT 3
),
defensive AS (
    SELECT
        p.name AS defensive_pokemon,
        ROW_NUMBER() OVER (ORDER BY AVG(pokemon_type_effectiveness(mm.type, p.type1, p.type2)) ASC) AS rn
    FROM trainer t
    JOIN pc ON t.trainer_id = pc.trainer_id
    JOIN pokedex p ON pc.pokemon_id = p.pokemon_id
    JOIN moves mm ON mm.name IN ('Hydro Pump', 'Dragon Dance', 'Dragon Pulse', 'Blizzard', 'Swift', 'Aqua Jet', 'Surf', 'Recover')
                 AND mm.power IS NOT NULL
    WHERE t.name = 'Red'
    GROUP BY pc.catch_id, p.name
    LIMIT 3
)
SELECT
    offensive.offensive_pokemon,
    defensive.defensive_pokemon
FROM offensive
JOIN defensive ON offensive.rn = defensive.rn;
